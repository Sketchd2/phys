use phys::math::Vec3;
fn load(path: &str) -> Vec<(f64, Vec<(Vec3, usize)>, Vec<(Vec3, usize)>)> {
    let mut out = vec![];
    for l in std::fs::read_to_string(path).unwrap().lines() {
        if l.starts_with('#') || l.trim().is_empty() { continue; }
        let (head, tail) = l.split_once('|').unwrap();
        let e: f64 = head.split_whitespace().nth(2).unwrap().parse().unwrap();
        let v: Vec<f64> = tail.split_whitespace().map(|x| x.parse().unwrap()).collect();
        let at = |i: usize| (Vec3 { x: v[4 * i], y: v[4 * i + 1], z: v[4 * i + 2] }, v[4 * i + 3] as usize);
        out.push((e * 627.509474, (0..3).map(at).collect(), (3..6).map(at).collect()));
    }
    out
}
fn report(label: &str, file: &str, law_file: &str) {
    let text = std::fs::read_to_string(law_file).unwrap();
    let law = phys::liquid::SiteSite::from_text(&text).unwrap();
    let alpha = phys::liquid::SiteSite::alpha_from_text(&text, law.charge.len());
    let bis = phys::liquid::SiteSite::bisector_from_text(&text);
    let extra = phys::liquid::extra_sites_from_text(&text);
    let sites = |m: &[(Vec3, usize)]| {
        let mut v = m.to_vec();
        if let Some((t, d)) = bis { v = phys::liquid::with_bisector_site(&v, t, d); }
        phys::liquid::with_frame_sites(&v, &extra)
    };
    let (mut n, mut s2, mut s3, mut n3) = (0, 0.0, 0.0, 0);
    let mut by_asym: Vec<(f64, f64)> = vec![];
    for (e, a, b) in load(file) {
        let pe = phys::liquid::PairEnergy { a: sites(&a), b: sites(&b), energy: 0.0 };
        let m = phys::liquid::pair_energy(&phys::liquid::Polarisable { law: &law, alpha: alpha.clone() }, &pe) * 627.509474;
        let d = m - e;
        n += 1; s2 += d * d;
        if e < 3.0 { n3 += 1; s3 += d * d; }
        let r = |p: &[(Vec3, usize)]| ((p[1].0 - p[0].0).norm() - (p[2].0 - p[0].0).norm()).abs();
        by_asym.push((r(&a).max(r(&b)), d));
    }
    by_asym.sort_by(|x, y| x.0.partial_cmp(&y.0).unwrap());
    let third = by_asym.len() / 3;
    let rms = |v: &[(f64, f64)]| (v.iter().map(|x| x.1 * x.1).sum::<f64>() / v.len().max(1) as f64).sqrt();
    println!("{label}: {n} pairs, rms {:.3} kcal/mol, below +3 kcal/mol ({n3}) {:.3}; by bond difference thirds (low/mid/high): {:.3} {:.3} {:.3}", (s2 / n as f64).sqrt(), (s3 / n3.max(1) as f64).sqrt(), rms(&by_asym[..third]), rms(&by_asym[third..2 * third]), rms(&by_asym[2 * third..]));
}
#[test]
fn opt2_probe() {
    let d = "runs/e8c/";
    let law = format!("{d}avg/law-final.txt");
    report("option 1 law on its own pairs (rigid)", &format!("{d}avg/pairs-water-gpu-bias-mp2.txt"), &law);
    report("option 1 law on flex uniform pairs  ", &format!("{d}flex/pairs-water-gpu-flex-mp2.txt"), &law);
    report("option 1 law on flex biased pairs   ", &format!("{d}flex/pairs-water-gpu-bias-flex-mp2.txt"), &law);
}
